```js
// Spoiler rule: M = the highest watched episode. Episodes up to M are shown fully (gaps below M too), the next one
// (up to M+1) shows its title and thumbnail without the summary, and everything after it is hidden.
// A revealed episode or series, or protection turned off, counts as watched.

const toNumber = (n) => {
  if (typeof n === 'number') return n
  const s = String(n ?? '').trim()
  return s === '' ? NaN : Number(s)
}

export function watchedMax(watched = []) {
  let max = 0
  for (const w of watched) {
    const n = toNumber(w)
    if (Number.isFinite(n) && n > max) max = n
  }
  return max
}

export function episodeVisibility(n, { enabled = true, watchedMax: max = 0, revealAll = false, revealed = [] } = {}) {
  if (!enabled || revealAll) return 'full'
  const num = toNumber(n)
  if (!Number.isFinite(num)) return 'hidden'
  if (num <= max || revealed.some((r) => toNumber(r) === num)) return 'full'
  if (num <= max + 1) return 'preview'
  return 'hidden'
}

// Built from a whitelist, never by spreading `meta`: a field ani.zip adds tomorrow cannot slip through.
// The image URL never leaves main; the renderer asks for the thumbnail by episode number.
export function redactEpisode(number, meta, visibility) {
  const runtime = Number.isFinite(meta?.runtime) && meta.runtime > 0 ? meta.runtime : null
  const out = { number: String(number), runtime, visibility }
  if (visibility === 'hidden' || !meta) return out
  if (typeof meta.title === 'string' && meta.title) out.title = meta.title
  out.hasThumb = typeof meta.image === 'string' && meta.image.length > 0
  if (visibility === 'full' && typeof meta.overview === 'string' && meta.overview) out.overview = meta.overview
  return out
}
```

```js
// ani-cli lists episodes by number ("1", "0", "12.5"); ani.zip keys its main episodes "1".."N".
// Only whole numbers from 1 get metadata; "0", decimals and specials fall back to "Episode N" + the series poster.

export function metaKey(label) {
  const s = String(label ?? '').trim()
  return /^[1-9]\d*$/.test(s) ? s : null
}

// More whole episodes at the source than ani.zip knows means the numbers cannot match (the source counts across
// seasons, or the series is matched to the wrong AniList entry): then no title may be shown, it could be a future one.
export function metaReliability(list, anizip) {
  const count = list.filter((l) => metaKey(l) != null).length
  const known = Number.isInteger(anizip?.episodeCount) && anizip.episodeCount > 0
    ? anizip.episodeCount
    : Object.keys(anizip?.episodes ?? {}).length
  return count <= known ? 'ok' : 'unreliable'
}

export function pickTitle(titles) {
  if (!titles || typeof titles !== 'object') return null
  for (const lang of ['en', 'x-jat']) {
    const v = titles[lang]
    if (typeof v === 'string' && v.trim()) return v.trim()
  }
  const any = Object.values(titles).find((v) => typeof v === 'string' && v.trim())
  return any ? any.trim() : null
}

export function mapEpisode(label, anizip) {
  const key = metaKey(label)
  const e = key ? anizip?.episodes?.[key] : null
  if (!e) return null
  return { title: pickTitle(e.titles), image: e.image ?? null, overview: e.overview ?? null, runtime: e.runtime ?? null }
}
```

```js
  return { all: r.all === true, episodes }
}

// Quality, sub/dub, the subtitle offset per mode and the spoiler reveals, remembered per series
// (also for series not in the watchlist).
export function createSeriesPrefs(file) {
  const loaded = readJson(file, { version: 1, series: {} }).data
  const db = loaded && typeof loaded.series === 'object' && loaded.series !== null ? loaded : { version: 1, series: {} }

  function get(title) {
    const raw = db.series[key(title)] ?? {}
    return {
      quality: QUALITIES.includes(raw.quality) ? raw.quality : null,
      mode: MODES.includes(raw.mode) ? raw.mode : null,
      subOffset: readOffsets(raw.subOffset),
    }
  }

  function getSpoilers(title) {
    const raw = db.series[key(title)] ?? {}
    return { reveal: readReveal(raw.reveal), metaOff: raw.metaOff === true }
  }

  // One writer for both halves, so changing one never drops the other.
  function persist(k, prefs, spoilers) {
    const stored = {}
    if (prefs.quality !== null) stored.quality = prefs.quality
    if (prefs.mode !== null) stored.mode = prefs.mode
    const offsets = Object.fromEntries(MODES.filter((m) => prefs.subOffset[m] !== 0).map((m) => [m, prefs.subOffset[m]]))
    if (Object.keys(offsets).length) stored.subOffset = offsets
    const { all, episodes } = spoilers.reveal
    if (all || episodes.length) stored.reveal = { ...(all ? { all: true } : {}), ...(episodes.length ? { episodes } : {}) }
    if (spoilers.metaOff) stored.metaOff = true
    if (Object.keys(stored).length) db.series[k] = stored
    else delete db.series[k]
    writeJsonAtomic(file, db)
  }

  function set(title, patch = {}) {
    const k = key(title)
    if (!k) return get(title)
    const next = get(title)
    if ('quality' in patch && (patch.quality === null || QUALITIES.includes(patch.quality))) next.quality = patch.quality
    if ('mode' in patch && (patch.mode === null || MODES.includes(patch.mode))) next.mode = patch.mode
    if (patch.subOffset !== null && typeof patch.subOffset === 'object') {
      for (const m of MODES) {
        if (!(m in patch.subOffset)) continue
        const v = patch.subOffset[m]
        if (v === null) next.subOffset[m] = 0
        else if (typeof v === 'number' && Number.isFinite(v)) next.subOffset[m] = clampOffset(v)
      }
    }
    persist(k, next, getSpoilers(title))
    return get(title)
  }

  function setSpoilers(title, action = {}) {
    const k = key(title)
    if (!k || action === null || typeof action !== 'object') return getSpoilers(title)
    const next = getSpoilers(title)
    let changed = false
    if ('episode' in action) {
      const ep = String(action.episode ?? '').trim()
      if (ep && !next.reveal.episodes.includes(ep)) { next.reveal.episodes.push(ep); changed = true }
    }
    if (action.all === true && !next.reveal.all) { next.reveal.all = true; changed = true }
    if (action.reset === true) { next.reveal = { all: false, episodes: [] }; changed = true }
    if (typeof action.metaOff === 'boolean' && action.metaOff !== next.metaOff) { next.metaOff = action.metaOff; changed = true }
    if (changed) persist(k, get(title), next)
    return getSpoilers(title)
  }

  function resolve(title, settings) {
    const p = get(title)
    return { quality: p.quality ?? settings.quality, mode: p.mode ?? settings.mode }
  }

  return { get, set, resolve, getSpoilers, setSpoilers }
}
```

```js
import path from 'node:path'
import { readJson, writeJsonAtomic } from './jsonStore.js'

export const ANIZIP_URL = 'https://api.ani.zip/mappings'
const HOUR = 3600 * 1000
export const AIRING_TTL_MS = 12 * HOUR
export const TTL_MS = 7 * 24 * HOUR
export const MISSING_TTL_MS = 7 * 24 * HOUR
export const TIMEOUT_MS = 15000

const MAIN_KEY = /^[1-9]\d*$/
const text = (v) => (typeof v === 'string' && v.trim() ? v.trim() : null)
export const validId = (id) => Number.isInteger(id) && id > 0

// Keep only what the app shows, so no other ani.zip field can ever reach the renderer.
export function slimAniZip(raw) {
  const episodes = {}
  const src = raw?.episodes !== null && typeof raw?.episodes === 'object' ? raw.episodes : {}
  for (const [k, e] of Object.entries(src)) {
    if (!MAIN_KEY.test(k) || e === null || typeof e !== 'object') continue
    const titles = {}
    for (const [lang, v] of Object.entries(e.title !== null && typeof e.title === 'object' ? e.title : {})) {
      if (text(v)) titles[lang] = text(v)
    }
    const runtime = Number(e.runtime ?? e.length)
    episodes[k] = {
      titles,
      image: typeof e.image === 'string' && e.image.startsWith('https://') ? e.image : null,
      overview: text(e.overview) ?? text(e.summary),
      runtime: Number.isFinite(runtime) && runtime > 0 ? Math.round(runtime) : null,
    }
  }
  const count = Number(raw?.episodeCount)
  return { episodeCount: Number.isInteger(count) && count > 0 ? count : Object.keys(episodes).length, episodes }
}

export function createEpisodeMeta({ cacheDir, fetchImpl = fetch, now = Date.now }) {
  const file = (id) => path.join(cacheDir, `${id}.json`)
  const inflight = new Map()

  async function fetchWithTimeout(url) {
    const ctrl = new AbortController()
    const timer = setTimeout(() => ctrl.abort(), TIMEOUT_MS)
    try { return await fetchImpl(url, { signal: ctrl.signal }) } finally { clearTimeout(timer) }
  }

  function readCache(id) {
    try {
      const d = readJson(file(id), null).data
      return d && Number.isFinite(d.fetchedAt) ? d : null
    } catch { return null }
  }

  async function load(id, airing) {
    const cached = readCache(id)
    const ttl = cached?.missing ? MISSING_TTL_MS : airing ? AIRING_TTL_MS : TTL_MS
    if (cached && now() - cached.fetchedAt < ttl) return cached.missing ? null : cached
    const stale = cached && !cached.missing ? cached : null
    try {
      const res = await fetchWithTimeout(`${ANIZIP_URL}?anilist_id=${id}`)
      if (res.status === 404) {
        writeJsonAtomic(file(id), { missing: true, fetchedAt: now() })
        return null
      }
      if (!res.ok) throw new Error(`ani.zip HTTP ${res.status}`)
      const data = { ...slimAniZip(await res.json()), fetchedAt: now() }
      writeJsonAtomic(file(id), data)
      return data
    } catch (err) {
      console.warn('episodeMeta: ani.zip request failed', err?.message)
      return stale
    }
  }

  function get(aniListId, { airing = false } = {}) {
    if (!validId(aniListId)) return Promise.resolve(null)
    if (!inflight.has(aniListId)) {
      inflight.set(aniListId, load(aniListId, airing).finally(() => inflight.delete(aniListId)))
    }
    return inflight.get(aniListId)
  }

  return { get, readCache }
}
```

```js
import { readJson, writeJsonAtomic } from './jsonStore.js'
import { normalizeTitle } from '../shared/domain.js'
import { MODES } from './settings.js'

const key = (title) => normalizeTitle(String(title ?? '')).toLowerCase()

// The real ani-cli episode list per series and mode ("0", "12.5", gaps), so the series page can show it
// instead of 1..max. Filled whenever ani-cli shows its episode menu.
export function createEpisodeLists(file, { now = () => new Date().toISOString() } = {}) {
  const loaded = readJson(file, { version: 1, series: {} }).data
  const db = loaded && typeof loaded.series === 'object' && loaded.series !== null ? loaded : { version: 1, series: {} }

  function get(title, mode) {
    const e = db.series[key(title)]?.[mode]
    if (!Array.isArray(e?.episodes)) return null
    const list = e.episodes.filter((x) => typeof x === 'string' && x)
    return list.length ? list : null
  }

  function set(title, mode, list) {
    const k = key(title)
    if (!k || !MODES.includes(mode) || !Array.isArray(list)) return
    const episodes = list.map((l) => String(l ?? '').trim()).filter(Boolean)
    if (!episodes.length) return
    const prev = get(title, mode)
    if (prev && prev.length === episodes.length && prev.every((x, i) => x === episodes[i])) return
    db.series[k] = { ...(db.series[k] ?? {}), [mode]: { episodes, at: now() } }
    writeJsonAtomic(file, db)
  }

  return { get, set }
}
```

```js
import { watchedMax, episodeVisibility, redactEpisode } from '../shared/spoilers.js'
import { metaKey, metaReliability, mapEpisode } from '../shared/episodeMap.js'

const AIRING = new Set(['RELEASING', 'NOT_YET_RELEASED'])
const NO_SPOILERS = { reveal: { all: false, episodes: [] }, metaOff: false }

// The one place that decides what the renderer may know about a series' episodes: every episode goes through
// redactEpisode here, so a hidden episode's title, picture or summary never leaves the main process.
export function createSeriesEpisodes({ library, seriesPrefs, settings, anilist, episodeMeta, episodeLists }) {
  async function resolve({ title = '', aniListId = null, mode = null, list = null } = {}) {
    const t = String(title ?? '').trim()
    const s = settings.get()
    const entry = t ? library.findByAniCliTitle(t) : null
    const watched = entry?.watchedEpisodes ?? []
    const listMode = mode === 'sub' || mode === 'dub' ? mode : t ? seriesPrefs.resolve(t, s).mode : s.mode

    let episodes
    let source
    const menu = Array.isArray(list) ? list.map((x) => String(x ?? '').trim()).filter(Boolean) : []
    if (menu.length) { episodes = menu; source = 'menu' } else {
      const cached = t ? episodeLists.get(t, listMode) : null
      if (cached?.length) { episodes = cached; source = 'ani-cli' } else {
        const max = Math.max(entry?.totalEpisodes ?? 0, ...watched.map(Math.ceil), 1)
        episodes = Array.from({ length: max }, (_, i) => String(i + 1))
        source = 'synthetic'
      }
    }

    const spoilers = t ? seriesPrefs.getSpoilers(t) : NO_SPOILERS
    const rule = { enabled: s.hideSpoilers, watchedMax: watchedMax(watched), revealAll: spoilers.reveal.all, revealed: spoilers.reveal.episodes }

    let meta = 'none'
    let anizip = null
    let id = null
    if (spoilers.metaOff) meta = 'off'
    else if (t || aniListId) {
      const knownId = aniListId ?? entry?.aniListId ?? null
      const info = await anilist.getForTitle(t, { aniListId: knownId }).catch(() => null)
      id = knownId ?? info?.id ?? null
      anizip = id ? await episodeMeta.get(id, { airing: AIRING.has(info?.status) }) : null
      if (anizip && Object.keys(anizip.episodes ?? {}).length) meta = source === 'synthetic' ? 'ok' : metaReliability(episodes, anizip)
    }
    return { episodes, source, meta, anizip, id, rule }
  }

  async function forSeries(args = {}) {
    const r = await resolve(args)
    return {
      source: r.source,
      meta: r.meta,
      episodes: r.episodes.map((n) => redactEpisode(n, r.meta === 'ok' ? mapEpisode(n, r.anizip) : null, episodeVisibility(n, r.rule))),
    }
  }

  // Checked again here, not trusted from the renderer: a hidden episode's picture is never even downloaded.
  async function thumb({ episode, ...args } = {}) {
    const n = String(episode ?? '').trim()
    const key = metaKey(n)
    if (!key) return null
    const r = await resolve(args)
    if (r.meta !== 'ok' || !r.episodes.includes(n)) return null
    if (episodeVisibility(n, r.rule) === 'hidden') return null
    return episodeMeta.thumb(r.id, key)
  }

  async function spoilers(args = {}, action = {}) {
    const t = String(args?.title ?? '').trim()
    if (t) seriesPrefs.setSpoilers(t, action)
    return forSeries(args)
  }

  return { forSeries, thumb, spoilers }
}
```

```js
import { useCallback, useEffect, useRef, useState } from 'react'
import { useApi } from '../api.js'
import { useLatest } from '../player/useLatest.js'

// The series' episode list with spoiler-safe metadata from main. Reloads when the library changes, because a newly
// watched episode moves the spoiler line. Only the latest request may set the state.
export function useSeriesEpisodes({ title, aniListId = null, mode = null, list = null }) {
  const api = useApi()
  const [series, setSeries] = useState(null)
  const args = useLatest({ title, aniListId, mode, list })
  const seq = useRef(0)
  const listKey = list ? list.join('\n') : ''

  const load = useCallback(() => {
    const n = ++seq.current
    if (!title) { setSeries(null); return }
    api.episodes.forSeries(args.current).then((r) => { if (n === seq.current) setSeries(r ?? null) }).catch(() => {})
  }, [api, title, aniListId, mode, listKey])

  useEffect(() => {
    load()
    return api.onLibraryChanged(load)
  }, [load])

  const update = useCallback(async (action) => {
    const n = ++seq.current
    const r = await api.episodes.spoilers(args.current, action).catch(() => null)
    if (r && n === seq.current) setSeries(r)
  }, [api])

  const thumb = useCallback((episode) => api.episodes.thumb({ ...args.current, episode }).catch(() => null), [api])

  return { series, update, thumb }
}
```

```js
import { useEffect, useRef, useState } from 'react'
import { useApi } from '../api.js'

// Numbers or list, remembered globally in settings.episodeView.
export function useEpisodeView() {
  const api = useApi()
  const [view, setView] = useState('numbers')
  const touched = useRef(false) // a click before the saved value arrives must win
  useEffect(() => {
    let live = true
    api.settings.get().then((s) => {
      if (live && !touched.current && (s?.episodeView === 'list' || s?.episodeView === 'numbers')) setView(s.episodeView)
    }).catch(() => {})
    return () => { live = false }
  }, [api])
  const change = (v) => {
    touched.current = true
    setView(v)
    api.settings.update({ episodeView: v }).catch(() => {})
  }
  return [view, change]
}
```

```jsx
import { useEffect, useRef, useState } from 'react'

// One episode in the list view. What it may show was decided in main: a hidden row only ever has number + runtime.
export function EpisodeRow({ row, poster = null, watched = false, selected = false, noted = false, onToggle, onReveal = null, loadThumb = null }) {
  const t = useT()
  const ref = useRef(null)
  const [src, setSrc] = useState(null)
  const [open, setOpen] = useState(false)
  const hidden = row.visibility === 'hidden'

  useEffect(() => {
    setSrc(null)
    if (hidden || !row.hasThumb || !loadThumb) return undefined
    let live = true
    const fetchThumb = () => loadThumb(row.number).then((u) => { if (live && u) setSrc(u) }).catch(() => {})
    // A group holds up to 100 rows: only ask for the pictures that scroll into view.
    if (typeof IntersectionObserver === 'undefined') {
      fetchThumb()
      return () => { live = false }
    }
    const io = new IntersectionObserver((entries) => {
      if (entries.some((e) => e.isIntersecting)) { io.disconnect(); fetchThumb() }
    })
    io.observe(ref.current)
    return () => { live = false; io.disconnect() }
  }, [row.number, hidden, row.hasThumb, loadThumb])

  const img = src ?? poster
  const label = row.title ? t('episodes.titled', { n: row.number, title: row.title }) : t('episodes.number', { n: row.number })
  const cls = ['ep-row', watched && 'ep-row--watched', selected && 'ep-row--selected', noted && 'ep-row--note', hidden && 'ep-row--hidden'].filter(Boolean).join(' ')
  return (
    <div ref={ref} className={cls}>
      <button type="button" className="ep-row__main" aria-pressed={selected} onClick={() => onToggle(row.number)}>
        {img ? <img className="ep-row__thumb" src={img} alt="" /> : <span className="ep-row__thumb ep-row__thumb--empty" aria-hidden="true" />}
        <span className="ep-row__text">
          <span className="ep-row__title">{label}</span>
          <span className="ep-row__meta muted">
            {row.runtime ? t('episodes.minutes', { n: row.runtime }) : null}
            {watched && <span className="ep-row__done">{t('episodes.watchedMark')}</span>}
          </span>
        </span>
      </button>
      {hidden && onReveal && <button type="button" onClick={() => onReveal(row.number)}>{t('episodes.reveal')}</button>}
      {row.overview && <button type="button" aria-expanded={open} onClick={() => setOpen((v) => !v)}>{t('episodes.summary')}</button>}
      {open && row.overview && <p className="ep-row__overview">{row.overview}</p>}
    </div>
  )
```
