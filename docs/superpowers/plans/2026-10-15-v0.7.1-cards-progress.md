**Goal:** One shared `MediaCard` (Home + Watchlist) with info on the poster, and the saved position of a partly watched episode shown on cards, Hero, episode pills and list rows.

**Architecture:** main `positions.js` gets `forSeries(title)` + `onChange(cb)`; `positionEvents.js` throttles changes into `event:positions-changed`; one IPC read channel. **Tech Stack:** Electron (main, ESM), React 18 (renderer), vitest + @testing-library/react (jsdom), plain CSS with tokens.

**Spec:** `docs/superpowers/specs/2026-10-15-v0.7.1-cards-progress-design.md`

## Review Focus

1. **Similar series titles** — "Naruto" must not pick up positions of "Naruto Shippuden" (key prefix includes `#`). Pinned in Task 1.
2. **Stale answers when the title changes fast** (Search switching series, Home re-sorting) — an older `forSeries` reply must never overwrite the newer one. Pinned in Task 2.
3. **Event storm while playing** — `save` every ~5 s must not trigger more than one renderer refresh per 5 s, but `clear` (episode just recorded as watched) must arrive immediately. Pinned in Task 1.

```js
import { EVENTS } from '../shared/channels.js'

// Trailing throttle: the first call opens a window, the window's end fires once.
export function createThrottle(fn, ms, { setTimer = setTimeout, clearTimer = clearTimeout } = {}) {
  let timer = null
  const run = () => { timer = null; fn() }
  return {
    call() { if (!timer) timer = setTimer(run, ms) },
    flush() { if (timer) clearTimer(timer); run() },
    dispose() { if (timer) clearTimer(timer); timer = null },
  }
}

// Positions are saved every ~5 s while playing: tell the renderer at most once per window. A clear (the episode
// was just recorded as watched) goes out at once so cards drop its "min left" right away.
export function connectPositionEvents({ positions, send, ms = 5000, setTimer, clearTimer }) {
  const throttle = createThrottle(() => send(EVENTS.positionsChanged), ms, { setTimer, clearTimer })
  const off = positions.onChange((kind) => (kind === 'clear' ? throttle.flush() : throttle.call()))
  return () => { off(); throttle.dispose() }
}
```

```js
import { useEffect, useState } from 'react'
import { useApi } from '../api.js'
import { progressRatio, remainingMinutes } from '../../shared/player.js'

const EMPTY = {}

function toProgress(raw) {
  const out = {}
  for (const [ep, v] of Object.entries(raw ?? {})) {
    if (!v || !Number.isFinite(v.position) || !Number.isFinite(v.duration) || v.duration <= 0) continue
    out[ep] = { position: v.position, duration: v.duration, ratio: progressRatio(v.position, v.duration), remaining: remainingMinutes(v.position, v.duration) }
  }
  return out
}

// Saved positions of one series (by ani-cli title), refreshed when main reports a change. Only numbers and times —
// never anything about what happens in an episode. Only the latest request may set the state.
export function useEpisodeProgress(title) {
  const api = useApi()
  const [progress, setProgress] = useState(EMPTY)
  useEffect(() => {
    setProgress(EMPTY)
    if (!title) return undefined
    let live = true
    let seq = 0
    const load = () => {
      const n = ++seq
      api.positions.forSeries(title)
        .then((r) => { if (live && n === seq) setProgress(toProgress(r)) })
        .catch(() => { if (live && n === seq) setProgress(EMPTY) })
    }
    load()
    const off = api.onPositionsChanged(load)
    return () => { live = false; off?.() }
  }, [api, title])
  return progress
}
```

```jsx
import { useEffect, useState } from 'react'
import { useApi } from '../api.js'
import { Icon } from './Icon.jsx'

// undefined = still loading, null = nothing found; callers that only read info?.poster see no difference.
export function usePosterInfo(title, aniListId = null) {
  const api = useApi()
  const [info, setInfo] = useState(title ? undefined : null)
  useEffect(() => {
    let live = true
    setInfo(title ? undefined : null)
    if (title) api.anilist.forTitle(title, aniListId).then((value) => { if (live) setInfo(value ?? null) }).catch(() => { if (live) setInfo(null) })
    return () => { live = false }
  }, [api, title, aniListId])
  return info
}

const Empty = () => <div className="poster poster--empty" aria-hidden="true"><Icon name="film" size={28} /></div>

// With children (a card overlay) the poster gets a frame and load states; without, the plain output stays as it was.
export function Poster({ title, aniListId = null, children = null }) {
  const info = usePosterInfo(title, aniListId)
  const src = info?.poster ?? null
  const [loaded, setLoaded] = useState(false)
  const [failed, setFailed] = useState(false)
  useEffect(() => { setLoaded(false); setFailed(false) }, [src])

  if (!children) return src ? <img className="poster" src={src} alt="" /> : <Empty />

  const missing = info === null || (info !== undefined && !src) || failed
  return (
    <div className="poster-frame">
      {missing ? <Empty /> : (
        <>
          {!loaded && <div className="poster poster--skeleton" aria-hidden="true" />}
          {src && <img className={loaded ? 'poster poster--in' : 'poster poster--loading'} src={src} alt="" onLoad={() => setLoaded(true)} onError={() => setFailed(true)} />}
        </>
      )}
      {children}
    </div>
  )
}
```

```jsx
import { useId } from 'react'
import { useT } from '../i18n/I18nContext.jsx'
import { Poster } from './Poster.jsx'
import { Icon } from './Icon.jsx'
import { nextEpisode } from '../../shared/domain.js'
import { useEpisodeProgress } from '../episodes/useEpisodeProgress.js'

export function isSeriesComplete(entry) {
  const total = entry.totalEpisodes
  if (!Number.isInteger(total) || total <= 0) return false
  for (let ep = 1; ep <= total; ep++) if (!entry.watchedEpisodes.includes(ep)) return false
  return true
}

// The line under the title. Only "EP N" — never an episode title (those only come through seriesEpisodes in main).
export function cardLine(entry, progress) {
  if (isSeriesComplete(entry)) return { key: 'card.completed', vars: {} }
  const next = nextEpisode(entry)
  const saved = progress[String(next)]
  if (saved) return { key: 'card.remaining', vars: { episode: next, min: saved.remaining } }
  if (entry.watchedEpisodes.length > 0 || entry.status === 'watching') return { key: 'card.next', vars: { episode: next } }
  return null
}

// Open and the actions are sibling buttons (a button cannot contain a button).
export function MediaCard({ entry, ready, onOpen, onContinue, onTogglePin = null, showScore = false }) {
  const t = useT()
  const lineId = useId()
  const key = entry.aniCliTitle ?? entry.title
  const progress = useEpisodeProgress(key)
  const total = entry.totalEpisodes
  const complete = isSeriesComplete(entry)
  const watched = Number.isInteger(total) && total > 0 ? Math.min(entry.watchedEpisodes.length, total) : entry.watchedEpisodes.length
  const ratio = Number.isInteger(total) && total > 0 ? Math.min(1, watched / total) : null
  const next = nextEpisode(entry)
  const line = cardLine(entry, progress)
  return (
    <div className="card media-card" data-status={entry.status} data-complete={complete ? 'true' : undefined}>
      <button
        type="button" className="media-card__open" onClick={() => onOpen(entry.id)}
        aria-label={t('card.label', { title: entry.title, watched, total: total ?? '?', status: t(`status.${entry.status}`) })}
        aria-describedby={line ? lineId : undefined}
      >
        <Poster title={entry.title} aniListId={entry.aniListId}>
          <span className="media-card__fade" aria-hidden="true" />
          {ratio != null && <span className="media-card__bar" aria-hidden="true" style={{ '--ratio': ratio }} />}
          <span className="media-card__status" aria-hidden="true" />
          <span className="badge badge--progress" aria-hidden="true">
            {complete ? `✓ ${total}/${total}` : <>{watched}<span className="muted">/{total ?? '?'}</span></>}
          </span>
          {showScore && entry.rating ? <span className="badge badge--score" aria-hidden="true">★ {entry.rating}</span> : null}
        </Poster>
        <span className="card__title">{entry.title}</span>
        {line && <span className="card__meta" id={lineId}>{t(line.key, line.vars)}</span>}
      </button>
      <div className="media-card__actions">
        {!complete && (
          <button type="button" className="primary media-card__play" disabled={!ready} aria-label={t('home.continueEp', { episode: next })} onClick={() => onContinue({ query: key, anime: key, episode: String(next) })}>
            <Icon name="play" />
          </button>
        )}
        {onTogglePin && (
          <button type="button" className="media-card__pin" aria-pressed={Boolean(entry.pinnedAt)} aria-label={`${entry.pinnedAt ? t('pin.remove') : t('pin.add')}: ${entry.title}`} onClick={() => onTogglePin(entry)}>
            <Icon name="star" />
          </button>
        )}
      </div>
    </div>
  )
}
```
