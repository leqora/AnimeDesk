**Goal:** Show how many episodes have aired since the last watched one ("N novih") and when the next one airs, and turn the series detail banner into a real header (meta, genres, actions).

**Tech Stack:** Electron main (Node, ESM via electron-vite), React 18, vitest + @testing-library/react (jsdom), plain CSS tokens.

**Spec:** `docs/superpowers/specs/2026-10-16-v0.7.2-airing-header-design.md`

```js
// Airing schedule logic. Only episode numbers and times ever come out of here — never an episode
// title, image or description (those only reach the renderer through main's seriesEpisodes.js).

// After an episode airs, AniList moves nextAiringEpisode on; give it an hour before refetching.
export const AIRED_GRACE_MS = 3600 * 1000

export const ANILIST_STATUSES = ['RELEASING', 'FINISHED', 'NOT_YET_RELEASED', 'HIATUS', 'CANCELLED']

// Self-correcting: once airingAt has passed, that episode counts as aired even if the cache is older.
export function airedCount(info, now) {
  const next = info?.nextAiring
  if (!next) return null
  return now >= next.airingAt ? next.episode : next.episode - 1
}

// "N new" = aired episodes after the highest watched one. Skipped episodes are not counted on purpose.
export function newEpisodeCount(entry, info, now) {
  if (entry?.status !== 'watching' || info?.status !== 'RELEASING') return 0
  const aired = airedCount(info, now)
  if (aired == null) return 0
  const maxWatched = Math.max(0, ...(entry.watchedEpisodes ?? []).filter(Number.isInteger))
  return Math.max(0, aired - maxWatched)
}

// After airingAt the next episode is unknown until the cache refreshes, so nothing is shown.
export function upcoming(info, now) {
  const next = info?.nextAiring
  return next && next.airingAt > now ? next : null
}

export function untilParts(ms) {
  if (!(ms >= 60_000)) return null
  const total = Math.floor(ms / 60_000)
  return { d: Math.floor(total / 1440), h: Math.floor((total % 1440) / 60), min: total % 60 }
}

export function untilKey(ms) {
  const p = untilParts(ms)
  if (!p) return { key: 'airing.soon', vars: {} }
  if (p.d >= 1) return p.h ? { key: 'airing.in.d', vars: { d: p.d, h: p.h } } : { key: 'airing.in.dOnly', vars: { d: p.d } }
  if (p.h >= 1) return p.min ? { key: 'airing.in.h', vars: { h: p.h, min: p.min } } : { key: 'airing.in.hOnly', vars: { h: p.h } }
  return { key: 'airing.in.min', vars: { min: p.min } }
}
```

```js
import sr from './sr.json'
import en from './en.json'

export const DICTS = { sr, en }
// Latin script for Serbian dates ("pet 17:30", not Cyrillic), 24-hour English.
export const LOCALES = { sr: 'sr-Latn', en: 'en-GB' }

export function createT(lang) {
  const code = DICTS[lang] ? lang : 'sr'
  const dict = DICTS[code]
  const t = (key, vars = {}) => {
    const s = dict[key] ?? DICTS.en[key] ?? key
    return s.replace(/\{(\w+)\}/g, (_, k) => (k in vars ? String(vars[k]) : `{${k}}`))
  }
  const rules = new Intl.PluralRules(code)
  // Keys come in _one/_few/_other in both languages; English uses one/other only.
  t.n = (key, count, vars = {}) => {
    const k = `${key}_${rules.select(count)}`
    return t(k in dict ? k : `${key}_other`, { count, ...vars })
  }
  t.lang = code
  t.locale = LOCALES[code]
  return t
}
```

```js
import { useEffect, useState } from 'react'

// A clock for countdowns; only the Hero and the detail page use it (cards compute once per render).
export function useNow(intervalMs = 60_000) {
  const [now, setNow] = useState(() => Date.now())
  useEffect(() => {
    const id = setInterval(() => setNow(Date.now()), intervalMs)
    return () => clearInterval(id)
  }, [intervalMs])
  return now
}
```

```jsx
// With children (a card overlay) the poster gets a frame and load states; without, the plain output stays as it was.
export function Poster({ title, aniListId = null, children = null }) {
  const info = usePosterInfo(title, aniListId)
  if (!children) {
    const src = info?.poster ?? null
    return src ? <img className="poster" src={src} alt="" /> : <Empty />
  }
  return <PosterFrame info={info}>{children}</PosterFrame>
}

// The framed poster for callers that already hold the AniList info (one request per card).
export function PosterFrame({ info, children }) {
  const src = info?.poster ?? null
  // Keyed by src so a new poster starts unloaded without a reset effect (which could race a fast onLoad).
  const [loadedSrc, setLoadedSrc] = useState(null)
  const [failedSrc, setFailedSrc] = useState(null)
  const loaded = src !== null && loadedSrc === src
  const failed = src !== null && failedSrc === src
  const missing = info === null || (info !== undefined && !src) || failed
  return (
    <div className="poster-frame">
      {missing ? <Empty /> : (
        <>
          {!loaded && <div className="poster poster--skeleton" aria-hidden="true" />}
          {src && <img className={loaded ? 'poster poster--in' : 'poster poster--loading'} src={src} alt="" onLoad={() => setLoadedSrc(src)} onError={() => setFailedSrc(src)} />}
        </>
      )}
      {children}
    </div>
  )
}
```
